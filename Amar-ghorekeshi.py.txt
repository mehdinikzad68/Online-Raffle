import pandas as pd

import matplotlib.pyplot as plt
import seaborn as sns
import os
import random
import pygame

# --- بخش گرافیکی قرعه‌کشی (GUI) ---
class WinnerAnimation(tk.Toplevel):
    """
    کلاسی برای ایجاد پنجره پاپ‌آپ انیمیشنی جهت نمایش نام برنده.
    """
    def __init__(self, parent, winner_name, winner_number, candidate_names):
        super().__init__(parent)
        self.winner_name = winner_name
        self.winner_number = winner_number
        self.candidate_names = candidate_names
        self.alpha = 0.0
        
        self.overrideredirect(True)
        screen_width = self.winfo_screenwidth()
        screen_height = self.winfo_screenheight()
        self.geometry(f"{screen_width}x{screen_height}+0+0")
        self.configure(bg='black')
        self.attributes('-alpha', self.alpha)
        
        self.shuffle_label = tk.Label(self, text="", font=("Arial", 100, "bold"), fg="white", bg="black")
        self.shuffle_label.pack(expand=True)
        
        self.play_sound()
        self.after(100, self.fade_in)

    def play_sound(self):
        try:
            pygame.mixer.init()
            pygame.mixer.music.load("draw_sound.mp3") 
            pygame.mixer.music.play()
        except pygame.error as e:
            print(f"هشدار: خطا در پخش فایل صوتی: {e}\nمطمئن شوید فایل 'draw_sound.mp3' کنار اسکریپت وجود دارد.")

    def fade_in(self):
        if self.alpha < 1.0:
            self.alpha += 0.05
            self.attributes('-alpha', self.alpha)
            self.after(50, self.fade_in)
        else:
            self.after(100, self.start_shuffling)

    def start_shuffling(self, shuffle_duration=5000, interval=100):
        random_name = random.choice(self.candidate_names)
        self.shuffle_label.config(text=random_name)
        if shuffle_duration > 0:
            self.after(interval, self.start_shuffling, shuffle_duration - interval, interval)
        else:
            pygame.mixer.music.stop()
            self.show_final_winner()

    def show_final_winner(self):
        final_text = f"برنده شماره {self.winner_number}\n\n{self.winner_name}"
        self.shuffle_label.config(text=final_text, fg="gold")
        self.font_size = 60
        self.zoom_in_text()

    def zoom_in_text(self):
        max_font_size = 120
        if self.font_size < max_font_size:
            self.font_size += 4
            self.shuffle_label.config(font=("Arial", self.font_size, "bold"))
            self.after(25, self.zoom_in_text)
        else:
            self.after(5000, self.close_window)

    def close_window(self):
        self.destroy()

class DrawWindow(tk.Toplevel):
    def __init__(self, parent, qualified_df, num_winners):
        super().__init__(parent)
        self.parent = parent
        self.qualified_df = qualified_df
        self.num_winners = num_winners
        self.winners_drawn = 0
        self.parent.winners_list = []

        self.title("پنجره قرعه کشی")
        self.geometry("600x500")

        input_frame = tk.Frame(self, pady=10)
        input_frame.pack(fill='x')
        tk.Label(input_frame, text="شماره ردیف برنده:", font=("Tahoma", 14)).pack(side='right', padx=10)
        self.entry = tk.Entry(input_frame, font=("Tahoma", 14), width=10, justify='center')
        self.entry.pack(side='right')
        self.draw_button = tk.Button(input_frame, text="انتخاب برنده", font=("Tahoma", 14, "bold"),
                                     bg="#4CAF50", fg="white", command=self.process_winner)
        self.draw_button.pack(side='left', padx=20)

        list_frame = tk.Frame(self, pady=10)
        list_frame.pack(fill='both', expand=True, padx=10)
        tk.Label(list_frame, text="لیست برندگان انتخاب شده", font=("Tahoma", 14, "bold")).pack()
        self.winners_listbox = tk.Listbox(list_frame, font=("Tahoma", 12), justify='right')
        self.winners_listbox.pack(fill='both', expand=True)

        self.protocol("WM_DELETE_WINDOW", self.on_closing)
        self.entry.focus_set()
        self.bind('<Return>', lambda event: self.process_winner())

    def process_winner(self):
        try:
            winner_num = int(self.entry.get())
        except ValueError:
            messagebox.showerror("خطا", "لطفاً یک شماره معتبر وارد کنید.", parent=self)
            return

        if winner_num not in self.qualified_df.index:
            messagebox.showwarning("خطا", "شماره وارد شده در لیست افراد واجد شرایط وجود ندارد.", parent=self)
            return

        if any(d['ردیف'] == winner_num for d in self.parent.winners_list):
            messagebox.showwarning("تکراری", "این شماره قبلاً به عنوان برنده انتخاب شده است.", parent=self)
            return

        winner_info = self.qualified_df.loc[winner_num]
        winner_name = winner_info['نام و نام خانوادگی']
        candidate_names = list(self.qualified_df['نام و نام خانوادگی'])
        
        self.withdraw()
        animation_window = WinnerAnimation(self, winner_name, self.winners_drawn + 1, candidate_names)
        self.wait_window(animation_window)
        self.deiconify()
        self.entry.delete(0, 'end')
        self.entry.focus_set()

        self.winners_drawn += 1
        winner_data = {
            'ردیف قرعه': self.winners_drawn, 'ردیف': winner_num,
            'نام و نام خانوادگی': winner_name, 'شماره تماس': winner_info.get('شماره تماس', 'N/A'),
            'ایمیل': winner_info.get('ایمیل', 'N/A'), 'امتیاز': winner_info['امتیاز']
        }
        self.parent.winners_list.append(winner_data)
        
        list_entry = f"{self.winners_drawn}. {winner_name} (ردیف شماره {winner_num})"
        self.winners_listbox.insert('end', list_entry)

        if self.winners_drawn >= self.num_winners:
            messagebox.showinfo("پایان قرعه‌کشی", f"قرعه‌کشی {self.num_winners} نفر به پایان رسید.", parent=self)
            self.draw_button.config(state='disabled', text="قرعه‌کشی تمام شد")

    def on_closing(self):
        if self.parent.winners_list:
            self.parent.save_winners_list()
        self.destroy()

# --- کلاس اصلی برنامه ---
class QuizSystem:
    def __init__(self, root):
        self.root = root
        self.root.withdraw()
        self.file_path = self.select_input_file()
        if not self.file_path:
            self.df = None
            return

        self.df = None
        self.qualified_df = None
        self.stats = {}
        self.winners_list = []
        self.load_and_analyze_data()

    def select_input_file(self):
        messagebox.showinfo("انتخاب فایل", "لطفاً فایل اکسل نتایج مسابقه را انتخاب کنید.")
        file_path = filedialog.askopenfilename(
            title="فایل اکسل نتایج را انتخاب کنید",
            filetypes=[("Excel Files", "*.xlsx *.xls")]
        )
        if not file_path:
            print("❌ هیچ فایلی انتخاب نشد. برنامه خاتمه می‌یابد.")
            return None
        return file_path

    def load_and_analyze_data(self):
        try:
            self.df = pd.read_excel(self.file_path)
            correct_answers = {
                'سوال 1': 'گزینه صحیح سوال 1',
                'سوال 2': 'گزینه صحیح سوال 2',
                'سوال 3': 'گزینه صحیح سوال 3'
            }
            def calculate_score(row):
                score = 0
                for q, a in correct_answers.items():
                    if q in row and str(row.get(q, '')).strip() == str(a).strip():
                        score += 1
                return score
            
            self.df['امتیاز'] = self.df.apply(calculate_score, axis=1)
            self.df = self.df.sort_values(by='امتیاز', ascending=False).reset_index(drop=True)
            self.qualified_df = self.df[self.df['امتیاز'] >= 2].reset_index(drop=True)
            self.qualified_df.index += 1
            
            self.stats = {
                'total': len(self.df), 'correct_3': len(self.df[self.df['امتیاز'] == 3]),
                'correct_2': len(self.df[self.df['امتیاز'] == 2]), 'correct_1': len(self.df[self.df['امتیاز'] == 1]),
                'correct_0': len(self.df[self.df['امتیاز'] == 0]), 'qualified_count': len(self.qualified_df)
            }
            print("✅ فایل اکسل با موفقیت خوانده و تحلیل شد.")
        except Exception as e:
            print(f"❌ خطایی در بارگذاری یا تحلیل فایل رخ داد: {e}")
            self.df = None

    def main_menu(self):
        while True:
            print("\n" + "="*40)
            print("   منوی اصلی سیستم مدیریت مسابقه همایش   ")
            print("="*40)
            print("1. نمایش آمار کلی شرکت‌کنندگان")
            print("2. نمایش لیست افراد واجد شرایط قرعه‌کشی")
            print("3. خروجی گزارش اکسل پیشرفته")
            print("4. نمایش نمودارهای گرافیکی")
            print("5. شروع فرآیند قرعه‌کشی")
            print("0. خروج از برنامه")
            print("="*40)
            
            choice = input("لطفاً گزینه مورد نظر را انتخاب کنید: ")
            
            if choice == '1': self.show_stats()
            elif choice == '2': self.show_qualified_list()
            elif choice == '3': self.export_to_excel()
            elif choice == '4': self.show_charts()
            elif choice == '5': self.start_draw_process()
            elif choice == '0':
                print("با تشکر از استفاده شما. خدانگهدار!")
                break
            else:
                print("❌ انتخاب نامعتبر است. لطفاً دوباره تلاش کنید.")

    def show_stats(self):
        print("\n--- آمار کلی مسابقه ---")
        for key, value in self.stats.items():
            print(f"{key.replace('_', ' ').title()}: {value}")
        input("\nبرای بازگشت به منوی اصلی، کلید Enter را فشار دهید...")

    def show_qualified_list(self):
        print("\n--- لیست افراد واجد شرایط قرعه‌کشی ---")
        if not self.qualified_df.empty:
            columns_to_show = ['نام و نام خانوادگی', 'شماره تماس', 'ایمیل', 'امتیاز']
            existing_cols = [col for col in columns_to_show if col in self.qualified_df.columns]
            print(self.qualified_df[existing_cols])
        else:
            print("هیچ فردی واجد شرایط نیست.")
        input("\nبرای بازگشت به منوی اصلی، کلید Enter را فشار دهید...")

    def export_to_excel(self):
        save_path = filedialog.asksaveasfilename(
            title="فایل گزارش اکسل را ذخیره کنید", defaultextension=".xlsx",
            filetypes=[("Excel Files", "*.xlsx")], initialfile="Quiz_Report.xlsx"
        )
        if not save_path:
            print("❌ عملیات ذخیره لغو شد.")
            return

        try:
            with pd.ExcelWriter(save_path, engine='xlsxwriter') as writer:
                stats_df = pd.DataFrame(list(self.stats.items()), columns=['آمار', 'مقدار'])
                stats_df.to_excel(writer, sheet_name='Dashboard', index=False)
                self.df.to_excel(writer, sheet_name='Full_Results', index=False)
                if not self.qualified_df.empty:
                    self.qualified_df.to_excel(writer, sheet_name='Qualified_List', index_label='ردیف')
            print(f"✅ گزارش کامل اکسل در مسیر '{save_path}' با موفقیت ذخیره شد.")
        except Exception as e:
            print(f"❌ خطایی در ذخیره فایل اکسل رخ داد: {e}")
        input("\nبرای بازگشت به منوی اصلی، کلید Enter را فشار دهید...")

    def show_charts(self):
        plt.style.use('seaborn-v0_8-darkgrid')
        plt.rcParams['font.family'] = 'Tahoma'
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(16, 7))
        pie_data = [self.stats['correct_3'], self.stats['correct_2'], self.stats['correct_1'], self.stats['correct_0']]
        pie_labels = ['3 پاسخ صحیح', '2 پاسخ صحیح', '1 پاسخ صحیح', '0 پاسخ صحیح']
        ax1.pie(pie_data, labels=pie_labels, autopct='%.1f%%', startangle=90, shadow=True, explode=(0.1, 0, 0, 0))
        ax1.set_title('توزیع امتیازات')
        bar_data = [self.stats['qualified_count'], self.stats['total'] - self.stats['qualified_count']]
        bar_labels = ['واجد شرایط', 'واجد شرایط نیستند']
        ax2.bar(bar_labels, bar_data, color=['#4CAF50', '#F44336'])
        ax2.set_title('وضعیت شرکت‌کنندگان برای قرعه‌کشی')
        print("در حال نمایش نمودارها... لطفاً پنجره نمودار را ببندید تا به منو بازگردید.")
        plt.show()

    def start_draw_process(self):
        if self.qualified_df.empty:
            messagebox.showerror("خطا", "هیچ فردی برای قرعه‌کشی واجد شرایط نیست.", parent=self.root)
            return

        num_winners = simpledialog.askinteger("تعداد برندگان", "تعداد برندگان مورد نظر را وارد کنید:", parent=self.root, minvalue=1, maxvalue=self.stats['qualified_count'])
        if not num_winners:
            print("❌ عملیات قرعه‌کشی لغو شد.")
            return

        draw_app = DrawWindow(self, self.qualified_df, num_winners)
        print("پنجره قرعه‌کشی باز شد. لطفاً عملیات را از طریق پنجره گرافیکی ادامه دهید...")
        self.root.wait_window(draw_app)
        
        print("\n✅ فرآیند قرعه‌کشی در پنجره گرافیکی به پایان رسید.")
        input("برای بازگشت به منوی اصلی، کلید Enter را فشار دهید...")

    def save_winners_list(self):
        if not self.winners_list: return
        save_choice = messagebox.askyesno("ذخیره لیست برندگان", "آیا مایلید لیست برندگان را در یک فایل اکسل ذخیره کنید؟", parent=self.root)
        if save_choice:
            save_path = filedialog.asksaveasfilename(
                title="فایل لیست برندگان را ذخیره کنید", defaultextension=".xlsx",
                filetypes=[("Excel Files", "*.xlsx")], initialfile="Winners_List.xlsx"
            )
            if save_path:
                try:
                    winners_df = pd.DataFrame(self.winners_list)
                    winners_df.to_excel(save_path, index=False)
                    print(f"✅ لیست برندگان با موفقیت در مسیر '{save_path}' ذخیره شد.")
                except Exception as e:
                    print(f"❌ خطایی در ذخیره فایل لیست برندگان رخ داد: {e}")
            else:
                print("❌ عملیات ذخیره لیست برندگان لغو شد.")

if __name__ == "__main__":
    root = tk.Tk()
    app = QuizSystem(root)
    # پنجره اصلی tkinter دیگر لازم نیست، پس آن را از بین می‌بریم
    root.destroy() 
    
    if app.df is not None:
        app.main_menu()
